# Rules for the plant monitor. One rule per line.
#
# Format:  <number>: <field> <operator> <threshold>
# The number is the buzz count: rule 2 buzzes twice.
#
# Fields:    air_humidity (%), air_temperature (C), soil_humidity (0-1000)
# Operators: <  >  <=  >=
#
# V0 and V1 read this file at runtime, on the Pi. V2 and V3 evaluate the rules
# on the ESP32, which cannot read the Pi's disk while the Pi is powered off, so
# the Pi pushes these lines over UART2 whenever the ESP32 is awake and the
# ESP32 keeps them in RTC memory. Press EN to apply a change immediately; in V3
# a change also reaches the ESP32 at its next wake. No reflash needed.
#
# install.sh still bakes the current lines into the firmware as a fallback, so
# a board that has never been pushed to, which means a power cycle, starts from
# them.
#
# These thresholds are placeholders. Calibrate them for your species and your
# probe: run tools/live.py --csv, note the raw adc in dry soil and again in a
# glass of water, then set SOIL_ADC_DRY and SOIL_ADC_WET in
# firmware/main/config.h to match.

1: air_humidity > 50
2: air_temperature < 15
3: soil_humidity < 300

# ---------------------------------------------------------------------------
# Timing. Optional: anything missing falls back to the variant's compiled
# default, so an empty file still runs. All values are in seconds.
#
#   sample_interval   how often the ESP32 samples
#   image_interval    how often the Pi takes a picture (V1)
#   heartbeat         how often the Pi photographs when nothing happens (V2)
#   adaptive_fast     interval while the readings are moving (V2, V3)
#   adaptive_slow     interval after three quiet samples (V2, V3)
#   wake_every_n_samples
#                     wake the Pi every N samples, rules ignored (V3). 0 means
#                     only when a rule fires, which is the deployed behaviour.
#                     For measurement and demos, 2 gives the cycle "sample
#                     alone, sample and wake the Pi, Pi logs and halts,
#                     repeat". Waking the Pi for nothing is exactly the cost V3
#                     exists to avoid, so leave it at 0 in the field.
#   wake_heartbeat    wake the Pi at least this often even if no rule fires
#                     (V3). 0 turns it off. A day is the point: without it a
#                     quiet plant and a dead node look the same in the log,
#                     because silence could mean either. One wake costs 92 J,
#                     which is 0.08 % of V3's daily energy.
#   log_max_lines     trim events_log.txt to this many lines. 0 = never trim,
#                     which is the default here: the log is the evidence for
#                     the report, so nothing is thrown away.
#
# The Pi pushes sample_interval, the adaptive bounds, wake_every_n_samples and
# wake_heartbeat over UART2, but only while the ESP32 is awake. In V3 that means
# setup mode, or the next time it wakes the Pi. Until then the values compiled
# into the variant apply.
# ---------------------------------------------------------------------------

sample_interval = 60
image_interval = 300
heartbeat = 3600
adaptive_fast = 10
adaptive_slow = 600
wake_heartbeat = 86400
wake_every_n_samples = 0
log_max_lines = 0
