# Measurement config for v3_loop.
#
# Copy this over config/configuration.txt before the run:
#     cp measurement/configs/v3_loop.txt config/configuration.txt
#
# The whole V3 cycle, repeated, so one capture shows it more than once:
#
#     ESP32 asleep, Pi off
#     ESP32 wakes, samples, sleeps          <- the Pi never sees this one
#     ESP32 wakes, samples, pulses the Pi
#     Pi boots, pulls the backlog, photographs, halts
#     repeat
#
# wake_every_n_samples = 2 is what produces that: the ESP32 pulses the Pi on
# every second sample regardless of the rules. 0 is the deployed behaviour,
# where only a rule fires a wake.
#
# The rules are quiet, so every wake in the trace comes from the counter and
# nothing else. The buzzer stays silent for the same reason: a beep every cycle
# would land in the measurement.
#
# COMPRESSED TIMESCALE. sample_interval is 20 s, not the deployed 60 s, so a
# 180 s capture holds two or three cycles instead of one. The energy per hour
# here is therefore NOT the deployed figure: it is the idle cost plus a wake
# every 2 x 20 s. Divide the event part by the number of wakes to get the cost
# of one, then compose for the real interval:
#
#     E/hour(real) = idle + events_per_hour(real) x event_cost
#
# adaptive_slow is pinned to adaptive_fast on purpose: left at 600 s the
# interval would stretch after three quiet samples and the cycle would stop.
#
# Deployed: 60 s fast / 600 s slow, camera per wake

1: air_humidity > 99
2: air_temperature < 0
3: soil_humidity < 0

# Timing, in seconds.
sample_interval = 20
adaptive_fast = 20
adaptive_slow = 20
wake_every_n_samples = 2
log_max_lines = 0
